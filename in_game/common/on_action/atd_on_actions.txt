# Monthly pulse for re-sending recalled diplomats
cmf_monthly_human_country_pulse = {
	on_actions = {
		atd_on_monthly
	}
}

# Monthly tick. Pre-populates atd_sync_check_list (the set of countries the GUI
# iter widgets walk) and sets atd_pending_action - the gate variable that flips
# the hidden window visible. The hidden window's single driver tick fires the
# iter trigger first, populating atd_improving_seen synchronously from the
# diplomatic engine, then runs reconcile + mod logic in atd_gui_sync_step.
# Pre-populating the list here means the GUI's datamodel has entries the moment
# the gate flips visible, so the iter widgets are instantiated and ready by the
# time the driver fires 0.5s later. The whole chain runs in one tick per month.
atd_on_monthly = {
	effect = {
		atd_update_hre_visibility = yes
		atd_gui_sync_begin = yes
		set_variable = atd_pending_action
	}
}
