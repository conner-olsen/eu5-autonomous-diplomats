atd_improve_relations = {
	save_scope_as = actor_scope
	set_variable = { name = atd_action_threshold value = { value = var:atd_min_diplomats add = 1 } }
	if = {
		limit = {
			num_of_diplomats >= var:atd_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
		}
		atd_update_improved_with = yes
		# Pass 1: prefer targets below +200 opinion
		set_variable = { name = atd_skip_high_relations value = 1 }
		cmm_for_each_list_item = {
			setting = atd__improve_relations
			effect = atd_improve_relations_each_item
		}
		# Pass 2: fallback to any remaining targets
		remove_variable = atd_skip_high_relations
		cmm_for_each_list_item = {
			setting = atd__improve_relations
			effect = atd_improve_relations_each_item
		}
	}
}

# Runs ONCE per monthly tick. Recomputes the per-type active counters and decays
# atd_improved_with for entries no longer being improved (per GUI sync).
# Active state and progress values are populated by the hidden window's GUI sync;
# this effect only handles the script-side bookkeeping that needs to run on the
# monthly cadence.
atd_update_improved_with = {
	set_variable = { name = atd_n_neighbors_improving_with value = 0 }
	set_variable = { name = atd_n_subjects_improving_with value = 0 }
	set_variable = { name = atd_n_outraged_improving_with value = 0 }
	set_variable = { name = atd_n_allies_improving_with value = 0 }
	set_variable = { name = atd_n_threatened_improving_with value = 0 }
	set_variable = { name = atd_n_hre_members_improving_with value = 0 }
	set_variable = { name = atd_n_hre_electors_improving_with value = 0 }

	if = {
		limit = {
			has_variable_map = atd_improved_with
		}
		every_key_in_variable_map = {
			variable = atd_improved_with
			save_scope_as = nation_improved_with
			scope:actor_scope = {
				if = {
					limit = {
						has_variable_map = atd_improving_with
						is_key_in_variable_map = {
							name = atd_improving_with
							target = scope:nation_improved_with
						}
					}
					# Active branch: country is currently being improved (per GUI).
					# Increment the per-type counter for entries the mod owns.
					# Player-initiated entries have no atd_improving_type and fall
					# through the switch without incrementing anything, by design.
					switch = {
						trigger = "variable_map(atd_improving_type|scope:nation_improved_with)"
						flag:neighbor = {
							change_variable = { name = atd_n_neighbors_improving_with add = 1 }
						}
						flag:subject = {
							change_variable = { name = atd_n_subjects_improving_with add = 1 }
						}
						flag:outraged = {
							change_variable = { name = atd_n_outraged_improving_with add = 1 }
						}
						flag:ally = {
							change_variable = { name = atd_n_allies_improving_with add = 1 }
						}
						flag:threatened = {
							change_variable = { name = atd_n_threatened_improving_with add = 1 }
						}
						flag:hre_member = {
							change_variable = { name = atd_n_hre_members_improving_with add = 1 }
						}
						flag:hre_elector = {
							change_variable = { name = atd_n_hre_electors_improving_with add = 1 }
						}
					}
				}
				else = {
					# Inactive branch: country is not currently being improved.
					# Decay the cooldown value by 0.1/month and drop below 1.
					set_variable = {
						name = temp_new_improved_amount
						value = {
							value = "variable_map(atd_improved_with|scope:nation_improved_with)"
							subtract = 0.1
						}
					}
					cmf_change_variable_map = {
						name = atd_improved_with
						key = scope:nation_improved_with
						value = var:temp_new_improved_amount
					}
					if = {
						limit = { "variable_map(atd_improved_with|scope:nation_improved_with)" < 1 }
						remove_from_variable_map = {
							name = atd_improved_with
							key = scope:nation_improved_with
						}
					}
				}
			}
		}
	}
}

# Begin phase of GUI sync. Builds atd_sync_check_list: every country the player
# could currently be improving with. Iter widgets walk this list and fire only
# for entries where IsImprovingOpinion is true, which gives us ground truth for
# both mod-initiated and player-initiated improve_relations actions. Also
# clears the per-cycle "seen" marker. Called by atd_gui_sync_step (cycle 1).
# Scope: player country.
#
# Scoping note: every_* iterators switch the body scope to the iterated item.
# Variable list/map operations apply to the current scope, so we have to switch
# back to the player country (scope:atd_sync_actor) before mutating player-owned
# storage. Same pattern in atd_gui_sync_end.
atd_gui_sync_begin = {
	save_scope_as = atd_sync_actor
	if = {
		limit = { has_variable_list = atd_improving_seen }
		clear_variable_list = atd_improving_seen
	}
	if = {
		limit = { has_variable_list = atd_sync_check_list }
		clear_variable_list = atd_sync_check_list
	}
	# Diplomatic range covers everyone the player can normally start an
	# improve_relations action against.
	every_country_in_diplomatic_range = {
		save_scope_as = atd_iter_country
		scope:atd_sync_actor = {
			add_to_variable_list = {
				name = atd_sync_check_list
				target = scope:atd_iter_country
			}
		}
	}
	# Catch tracked entries that may have fallen out of diplomatic range mid-action.
	if = {
		limit = { has_variable_map = atd_improving_with }
		every_key_in_variable_map = {
			variable = atd_improving_with
			save_scope_as = atd_tracked_country
			scope:atd_sync_actor = {
				if = {
					limit = {
						NOT = {
							is_target_in_variable_list = {
								name = atd_sync_check_list
								target = scope:atd_tracked_country
							}
						}
					}
					add_to_variable_list = {
						name = atd_sync_check_list
						target = scope:atd_tracked_country
					}
				}
			}
		}
	}
}

# End phase of GUI sync. Reconciles atd_improving_with against atd_improving_seen:
#   - Entries in atd_improving_with not seen this pass (canceled or completed)
#     are dropped from atd_improving_with and atd_improving_type. atd_improved_with
#     retains its last seen progress value and decays via atd_update_improved_with.
#   - Entries in atd_improving_seen not yet in atd_improving_with are player-
#     initiated improvements the mod just discovered; add them to atd_improving_with.
#     atd_improving_type stays unset for these so they don't count toward mod budgets.
#
# Stale entries are collected into atd_improving_to_remove first, then processed in
# a separate pass so we never mutate atd_improving_with map keys during their own
# iteration. Called by atd_gui_sync_step (cycle 2). Scope: player country.
atd_gui_sync_end = {
	save_scope_as = atd_sync_actor
	if = {
		limit = { has_variable_list = atd_improving_to_remove }
		clear_variable_list = atd_improving_to_remove
	}
	if = {
		limit = { has_variable_map = atd_improving_with }
		every_key_in_variable_map = {
			variable = atd_improving_with
			save_scope_as = atd_nation_to_check
			scope:atd_sync_actor = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable_list = atd_improving_seen }
							NOT = {
								is_target_in_variable_list = {
									name = atd_improving_seen
									target = scope:atd_nation_to_check
								}
							}
						}
					}
					add_to_variable_list = {
						name = atd_improving_to_remove
						target = scope:atd_nation_to_check
					}
				}
			}
		}
	}
	if = {
		limit = { has_variable_list = atd_improving_to_remove }
		every_in_list = {
			variable = atd_improving_to_remove
			save_scope_as = atd_nation_to_remove
			scope:atd_sync_actor = {
				remove_from_variable_map = {
					name = atd_improving_with
					key = scope:atd_nation_to_remove
				}
				if = {
					limit = {
						has_variable_map = atd_improving_type
						is_key_in_variable_map = {
							name = atd_improving_type
							target = scope:atd_nation_to_remove
						}
					}
					remove_from_variable_map = {
						name = atd_improving_type
						key = scope:atd_nation_to_remove
					}
				}
			}
		}
		clear_variable_list = atd_improving_to_remove
	}
	# Add newly discovered improvements (typically player-initiated) to the map
	# so subsequent mod ticks correctly see them as in-progress and skip them.
	if = {
		limit = { has_variable_list = atd_improving_seen }
		every_in_list = {
			variable = atd_improving_seen
			save_scope_as = atd_nation_seen
			scope:atd_sync_actor = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable_map = atd_improving_with }
							NOT = {
								is_key_in_variable_map = {
									name = atd_improving_with
									target = scope:atd_nation_seen
								}
							}
						}
					}
					add_to_variable_map = {
						name = atd_improving_with
						key = scope:atd_nation_seen
						value = 1
					}
				}
			}
		}
		clear_variable_list = atd_improving_seen
	}
	if = {
		limit = { has_variable_list = atd_sync_check_list }
		clear_variable_list = atd_sync_check_list
	}
}

atd_improve_relations_each_item = {
	# $i$ is the resolved item number (1-7)
	# scope:cmm_list_current_item_value  (attached game object)
	# Per-item aliases for count:
	#   Neighboring Countries: var:atd_improve_neighbors_count
	#   Own Subject Countries: var:atd_improve_relations_count
	#   Outraged Countries: var:atd_improve_outraged_count
	#   Allies: var:atd_improve_allies_count
	#   Threatening Countries: var:atd_improve_threatened_count
	#   HRE Members: var:atd_improve_hre_members_count
	#   HRE Electors: var:atd_improve_hre_electors_count

	switch = {
		trigger = scope:cmm_list_current_item_value
		flag:neighbors = { atd_improve_neighbors = yes }
		flag:subjects = { atd_improve_subjects = yes }
		flag:outraged = { atd_improve_outraged = yes }
		flag:allies = { atd_improve_allies = yes }
		flag:threatening = { atd_improve_threatened = yes }
		flag:hre_members = { atd_improve_hre_members = yes }
		flag:hre_electors = { atd_improve_hre_electors = yes }
	}

	cmf_suppress = { v = $i$ }
	cmf_suppress = { v = $setting$ }
}

# Starts improving relations with scope:target_country if not already improving.
# Must be called inside scope:actor_scope.
# Params: type (type flag name), counter (counter variable name), skip_threshold (max progress to skip at)
atd_start_improving_with = {
	if = {
		limit = {
			num_of_diplomats >= var:atd_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
			OR = {
				NOT = { has_variable_map = atd_improving_with }
				NOT = {
					is_key_in_variable_map = {
						name = atd_improving_with
						target = scope:target_country
					}
				}
			}
			OR = {
				NOT = { has_variable_map = atd_improved_with }
				NOT = {
					is_key_in_variable_map = {
						name = atd_improved_with
						target = scope:target_country
					}
				}
				"variable_map(atd_improved_with|scope:target_country)" < $skip_threshold$
			}
			trigger_if = {
				limit = { has_variable = atd_skip_high_relations }
				scope:target_country = {
					NOT = { opinion = { target = scope:actor_scope value >= 200 } }
				}
			}
		}
		perform_diplomatic_action = {
			type = improve_relation
			actor = scope:actor_scope
			recipient = scope:target_country
		}
		add_to_variable_map = {
			name = atd_improving_with
			key = scope:target_country
			value = 1
		}
		add_to_variable_map = {
			name = atd_improved_with
			key = scope:target_country
			value = 1
		}
		add_to_variable_map = {
			name = atd_improving_type
			key = scope:target_country
			value = flag:$type$
		}
		change_variable = { name = $counter$ add = 1 }
	}

	# Suppress "used but never set" warnings for type flags passed via $type$
	if = {
		limit = { always = no }
		cmf_suppress = { v = neighbor }
		cmf_suppress = { v = subject }
		cmf_suppress = { v = outraged }
		cmf_suppress = { v = ally }
		cmf_suppress = { v = threatened }
		cmf_suppress = { v = hre_member }
		cmf_suppress = { v = hre_elector }
	}
}

atd_improve_neighbors = {
	while = {
		limit = {
			var:atd_n_neighbors_improving_with < var:atd_improve_neighbors_count
			num_of_diplomats >= var:atd_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
		}
		count = 10
		random_neighbor_country = {
			save_scope_as = target_country
			scope:actor_scope = {
				atd_start_improving_with = { type = neighbor counter = atd_n_neighbors_improving_with skip_threshold = 90 }
			}
		}
	}
}

atd_improve_outraged = {
	while = {
		limit = {
			var:atd_n_outraged_improving_with < var:atd_improve_outraged_count
			num_of_diplomats >= var:atd_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
		}
		count = 10
		ordered_country_in_diplomatic_range = {
			limit = {
				is_discovered_by = scope:actor_scope
				antagonism = { target = scope:actor_scope value > 0 }
				scope:actor_scope = {
					OR = {
						NOT = { has_variable_map = atd_improving_with }
						NOT = {
							is_key_in_variable_map = {
								name = atd_improving_with
								target = PREV
							}
						}
					}
				}
			}
			order_by = this_antagonism_towards_prev
			max = 1
			save_scope_as = target_country
			scope:actor_scope = {
				atd_start_improving_with = { type = outraged counter = atd_n_outraged_improving_with skip_threshold = 90 }
			}
		}
	}
}

# Show/hide HRE items based on eligibility.
# Item 6 (HRE Electors): visible if you are an HRE member.
# Item 7 (HRE Members): visible if you are an HRE member.
# Scope: country
atd_update_hre_visibility = {
	if = {
		limit = {
			is_member_of_international_organization = international_organization:hre
		}
		cmm_show_list_item = {
			mod_id = atd
			setting_id = improve_relations
			item = 6
		}
	}
	else = {
		cmm_hide_list_item = {
			mod_id = atd
			setting_id = improve_relations
			item = 6
		}
	}
	if = {
		limit = {
			is_member_of_international_organization = international_organization:hre
		}
		cmm_show_list_item = {
			mod_id = atd
			setting_id = improve_relations
			item = 7
		}
	}
	else = {
		cmm_hide_list_item = {
			mod_id = atd
			setting_id = improve_relations
			item = 7
		}
	}
}
