# Autonomous diplomat effects — debug build (neighbors only).
# Uses variable lists (atd_targets, atd_decaying) on root instead of every_country scans.
# Scope: country (the player)

# Monthly tick: track improvement progress and decay.
atd_monthly_tick = {
	# Decrement countdown for all active targets
	every_in_list = {
		variable = atd_targets
		change_variable = { name = atd_months_remaining subtract = 1 }
	}

	# Clear targets that hit the opinion cap
	every_in_list = {
		variable = atd_targets
		limit = { var:atd_months_remaining <= 0 }
		save_scope_as = atd_finished
		remove_variable = atd_months_remaining
		remove_variable = atd_auto_target
		set_variable = { name = atd_room value = 0 }
		set_variable = { name = atd_decay_sub value = 0 }
		root = {
			remove_list_variable = { name = atd_targets target = scope:atd_finished }
			add_to_variable_list = { name = atd_decaying target = scope:atd_finished }
		}
	}

	# Tick decay: each month adds 1 to sub-counter, every 12 = 1 opinion point of room
	every_in_list = {
		variable = atd_decaying
		change_variable = { name = atd_decay_sub add = 1 }
		if = {
			limit = { var:atd_decay_sub >= 12 }
			change_variable = { name = atd_room add = 1 }
			change_variable = { name = atd_decay_sub subtract = 12 }
		}
		if = {
			limit = { var:atd_room >= 100 }
			save_scope_as = atd_decayed
			remove_variable = atd_room
			remove_variable = atd_decay_sub
			root = { remove_list_variable = { name = atd_decaying target = scope:atd_decayed } }
		}
	}
}

# Count existing neighbor assignments.
atd_count_assignments = {
	atd_monthly_tick = yes
	set_variable = { name = atd_assigned_neighbors value = 0 }
	every_in_list = {
		variable = atd_targets
		limit = { var:atd_auto_target = 2 }
		root = { change_variable = { name = atd_assigned_neighbors add = 1 } }
	}
}

# Master effect — neighbors only for debugging.
atd_auto_improve = {
	atd_count_assignments = yes
	atd_improve_neighbors = yes
	remove_variable = atd_assigned_neighbors
}

# Stub dispatchers so cmm_for_each_list_item doesn't break.
atd_dispatch_item = { cmf_suppress = { v = $setting$ } }
atd_improve_subjects = { }
atd_improve_outraged = { }
atd_improve_allies = { }
atd_improve_threatened = { }

# ── Neighbors (atd_auto_target = 2) ─────────────────────────────────────────
atd_improve_neighbors = {
	if = {
		limit = {
			has_variable = atd_improve_neighbors_count
			var:atd_improve_neighbors_count > 0
		}
		set_variable = { name = atd_remaining value = var:atd_improve_neighbors_count }
		change_variable = { name = atd_remaining subtract = var:atd_assigned_neighbors }

		# Recall excess targets
		while = {
			count = 10
			limit = {
				var:atd_remaining < 0
				any_country = { has_variable = atd_auto_target var:atd_auto_target = 2 }
			}
			random_country = {
				limit = { has_variable = atd_auto_target var:atd_auto_target = 2 }
				save_scope_as = atd_recall_target
				if = {
					limit = { has_variable = atd_months_remaining }
					set_variable = { name = atd_room value = var:atd_months_remaining }
					remove_variable = atd_months_remaining
				}
				remove_variable = atd_auto_target
				set_variable = { name = atd_decay_sub value = 0 }
			}
			remove_list_variable = { name = atd_targets target = scope:atd_recall_target }
			add_to_variable_list = { name = atd_decaying target = scope:atd_recall_target }
			change_variable = { name = atd_remaining add = 1 }
		}

		# Dispatch new targets
		while = {
			count = 10
			limit = {
				var:atd_remaining > 0
				num_of_diplomats > var:atd_min_diplomats
				modifier:monthly_diplomats >= var:atd_min_diplomat_gain
				any_neighbor_country = {
					is_discovered_by = root
					NOT = { has_variable = atd_auto_target }
					NOT = { is_subject_of = root }
					OR = {
						NOT = { has_variable = atd_decay_sub }
						var:atd_room >= 12
					}
				}
			}
			random_neighbor_country = {
				limit = {
					is_discovered_by = root
					NOT = { has_variable = atd_auto_target }
					NOT = { is_subject_of = root }
					OR = {
						NOT = { has_variable = atd_decay_sub }
						var:atd_room >= 12
					}
				}
				save_scope_as = atd_send_target
				set_variable = { name = atd_auto_target value = 2 }
				set_variable = { name = atd_months_remaining value = 100 }
				if = {
					limit = { has_variable = atd_room }
					set_variable = { name = atd_months_remaining value = var:atd_room }
				}
				if = {
					limit = { has_variable = atd_decay_sub }
					remove_variable = atd_decay_sub
				}
			}
			remove_list_variable = { name = atd_decaying target = scope:atd_send_target }
			add_to_variable_list = { name = atd_targets target = scope:atd_send_target }
			perform_diplomatic_action = {
				type = improve_relation
				actor = root
				recipient = scope:atd_send_target
			}
			change_variable = { name = atd_remaining subtract = 1 }
		}
		remove_variable = atd_remaining
	}
	else_if = {
		limit = {
			has_variable = atd_improve_neighbors_count
			var:atd_assigned_neighbors > 0
		}
		# Full recall of all neighbor targets
		every_in_list = {
			variable = atd_targets
			limit = { var:atd_auto_target = 2 }
			save_scope_as = atd_recalled
			if = {
				limit = { has_variable = atd_months_remaining }
				set_variable = { name = atd_room value = var:atd_months_remaining }
				remove_variable = atd_months_remaining
			}
			remove_variable = atd_auto_target
			set_variable = { name = atd_decay_sub value = 0 }
			root = {
				remove_list_variable = { name = atd_targets target = scope:atd_recalled }
				add_to_variable_list = { name = atd_decaying target = scope:atd_recalled }
			}
		}
	}
}
