# Driver step. Toggles between begin (cycle 1) and end (cycle 2) phases, and
# also runs deferred monthly mod logic so it sees fresh sync state.
#
# Cycle 1 (no flag): decrements atd_pending_action_countdown if monthly tick set
# it; clears the per-cycle "seen" marker; sets the collecting flag so the iter
# widgets become visible and fire tracking calls.
# Cycle 2 (flag set): reconciles via atd_gui_sync_end; if the countdown hit 0
# this cycle, runs atd_enforce_on_subjects + atd_improve_relations against the
# now-fresh atd_improving_with. Then clears the flag.
# Root: player country.
atd_gui_sync_step = {
	effect = {
		if = {
			limit = { has_variable = atd_gui_sync_collecting }
			atd_gui_sync_end = yes
			remove_variable = atd_gui_sync_collecting
			# Run deferred monthly logic if the begin phase brought the countdown
			# to 0 this cycle. This means a full begin -> iter -> end pass just
			# completed with the latest engine state, so atd_improving_with
			# reflects player cancellations and natural completions.
			if = {
				limit = {
					has_variable = atd_pending_action_countdown
					var:atd_pending_action_countdown <= 0
				}
				atd_enforce_on_subjects = yes
				atd_improve_relations = yes
				remove_variable = atd_pending_action_countdown
			}
		}
		else = {
			atd_gui_sync_begin = yes
			set_variable = atd_gui_sync_collecting
			if = {
				limit = { has_variable = atd_pending_action_countdown }
				change_variable = { name = atd_pending_action_countdown subtract = 1 }
			}
		}
	}
}

# Visibility gate for the iter section. Only true between begin and end phases.
# Root: player country.
atd_gui_sync_is_collecting = {
	is_shown = {
		has_variable = atd_gui_sync_collecting
	}
}

# Per-country tracking call. Fired once per ongoing improve_opinion country
# during the begin phase via PdxGuiTriggerAllAnimations. Refreshes the GUI
# progress value into atd_improved_with and adds the country to the seen list
# so the end phase knows this entry is still active.
# Root: player country. Saved scopes: country (target), progress (0-100).
atd_gui_sync_track = {
	saved_scopes = { country progress }
	effect = {
		cmf_change_variable_map = {
			name = atd_improved_with
			key = scope:country
			value = scope:progress
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable_list = atd_improving_seen }
					NOT = {
						is_target_in_variable_list = {
							name = atd_improving_seen
							target = scope:country
						}
					}
				}
			}
			add_to_variable_list = {
				name = atd_improving_seen
				target = scope:country
			}
		}
	}
}
