# Improve Relations (improve_relations) Tab
atd__improve_relations_on_changed = {
	scope = country

	effect = {
		cmm_apply_list_change = {
			setting = atd__improve_relations
		}
		cmm_sync_setting_alias = {
			setting = atd__improve_relations_i1_f1
			alias = atd_improve_neighbors_count
		}
		cmm_sync_setting_alias = {
			setting = atd__improve_relations_i2_f1
			alias = atd_improve_relations_count
		}
		cmm_sync_setting_alias = {
			setting = atd__improve_relations_i3_f1
			alias = atd_improve_outraged_count
		}
		cmm_sync_setting_alias = {
			setting = atd__improve_relations_i4_f1
			alias = atd_improve_allies_count
		}
		cmm_sync_setting_alias = {
			setting = atd__improve_relations_i5_f1
			alias = atd_improve_threatened_count
		}
		cmm_sync_setting_alias = {
			setting = atd__improve_relations_i6_f1
			alias = atd_improve_hre_electors_count
		}
		cmm_sync_setting_alias = {
			setting = atd__improve_relations_i7_f1
			alias = atd_improve_hre_members_count
		}
	}
}

# Subjects (subjects) Tab
atd__enforce_on_changed = {
	scope = country

	effect = {
		cmm_apply_list_change = {
			setting = atd__enforce
		}
		cmm_sync_bool_alias = {
			setting = atd__enforce_i1_f1
			alias = atd_enforce_religion_enabled
		}
		cmm_sync_bool_alias = {
			setting = atd__enforce_i2_f1
			alias = atd_enforce_culture_enabled
		}
	}
}

# Card controls in atd_automation_lateralview.gui.
# Root: player country.
atd_automation_enable_improve = {
	scope = country

	effect = {
		atd_set_bool_setting = { setting = atd__improve_relations_enabled value = 1 }
	}
}

atd_automation_disable_improve = {
	scope = country

	effect = {
		atd_set_bool_setting = { setting = atd__improve_relations_enabled value = 0 }
	}
}
